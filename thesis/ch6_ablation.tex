\section{Specification of Ablation Models M1--M4}
To evaluate the contribution of each algorithmic component without confounding factors, we adopt a single-factor ablation design. Four model configurations (M1 through M4) are specified such that each successive transition alters exactly one algorithmic mechanism while holding all other parameters, training budgets, random seeds, transaction costs, and data folds strictly invariant:
\begin{enumerate}
    \setlength{\itemsep}{1.5pt}
    \setlength{\parskip}{0pt}
    \item \textbf{Model M1 (MLP Baseline + Net Return):} Two-layer feedforward MLP observing solely instantaneous market observation $\mathbf{s}_t \in \mathbb{R}^{141}$ with net return reward ($r_t = R_{\text{net},t}$) and zero penalty ($\lambda_{\text{turn}} = 0$).
    \item \textbf{Model M2 (LSTM Recurrent Encoder + Net Return):} Replaces the static MLP with a single-layer LSTM ($H=512$) over sliding window $\mathbf{F}_t \in \mathbb{R}^{30 \times 141}$. Reward remains net return ($r_t = R_{\text{net},t}$). Transition M1 $\to$ M2 isolates \textbf{temporal memory and sequence modeling} \citep{zou2024novel}.
    \item \textbf{Model M3 (LSTM Encoder + Differential Sharpe Ratio):} Retains the LSTM backbone, replacing net return with the recursive Differential Sharpe Ratio ($r_t = D_t$). Transition M2 $\to$ M3 isolates \textbf{risk-aware reward engineering} \citep{millea2021deep}.
    \item \textbf{Model M4 (LSTM + DSR + Quadratic Turnover Regularisation):} Retains the LSTM backbone and DSR reward, augmented by a quadratic weight-change penalty on risky assets ($r_t = D_t - \lambda_{\text{turn}} \sum_{j=1}^{10} (w_{j,t}^{\text{target}} - w_{j,t}^{\text{cur}})^2$) with $\lambda_{\text{turn}} = 0.1$. Transition M3 $\to$ M4 isolates \textbf{turnover regularisation} \citep{wang2025risk}.
\end{enumerate}

\section{Control Invariants and Training Algorithm}
Experimental controls are strictly enforced: (i)~Actor and Critic heads are instantiated prior to the encoder to ensure identical initial head weights across models under the same seed; (ii)~all models train for a nominal budget of 300,000 environment interaction steps, executing exactly 2,344 rollout iterations of horizon $T_{\text{rollout}} = 128$ for a total of 300,032 completed transitions, with identical Adam learning rate ($3 \times 10^{-4}$), batch size (64), and 10 PPO epochs; (iii)~all models experience identical market sequences and transaction costs; and (iv)~programmatic assertions verify that each transition alters exactly one dimension. Algorithm~\ref{alg:ppo} details the complete training workflow.

\begin{algorithm}[H]
\caption{Continuous Dirichlet PPO Daily Portfolio Allocation}\label{alg:ppo}
\small
\begin{algorithmic}[1]
\Require Market indicators $\mathbf{X}$, returns $\mathbf{R}$, cost $c_{\text{trans}} = 0.001$, threshold $\tau = 0.01$, penalty $\lambda_{\text{turn}}$, DSR rate $\eta_{\text{dsr}} = 0.01$, PPO hyperparameters ($\gamma, \lambda_{\text{GAE}}, \epsilon, \eta = 3 \times 10^{-4}$).
\Ensure Optimised policy weights $\theta^*$ selected by peak validation Sharpe ratio.
\State \textbf{Initialise} shared Encoder $\psi$, Actor $\theta$, Critic $\phi$, Adam optimiser; $V_0 \gets 10^6$, $\mathbf{w}_0^{\text{cur}} \gets [0,\dots,0,1]^\top$; DSR moments $A_0 \gets 0, B_0 \gets 0$; step counter $N_{\text{step}} \gets 0, N_{\text{eval}} \gets 25{,}000$, $\text{SR}_{\text{best}} \gets -\infty$.
\While{$N_{\text{step}} < 300{,}000$} \Comment{Completes 2,344 rollouts $\times 128 = 300{,}032$ environment steps}
  \State Reset rollout buffer $\mathcal{B} \gets \emptyset$.
  \For{$t = 1$ \textbf{to} $T_{\text{rollout}} = 128$}
    \State Observe window $\mathbf{F}_t \in \mathbb{R}^{30 \times 141}$; extract instantaneous state $\mathbf{s}_t \in \mathbb{R}^{141}$ (last row of $\mathbf{F}_t$).
    \State Compute latent state $\mathbf{h}_t \gets \text{Encoder}_\psi(\mathbf{s}_t)$ for M1, or $\text{Encoder}_\psi(\mathbf{F}_t)$ for M2--M4.
    \State Compute raw logits $\mathbf{z}_t \gets \mathbf{W}_a \mathbf{h}_t + \mathbf{b}_a$ and concentration parameters $\boldsymbol{\alpha}_t \gets \exp(\mathbf{z}_t)$.
    \State Sample raw continuous action $\mathbf{a}_t \sim \operatorname{Dirichlet}(\boldsymbol{\alpha}_t)$ via Gamma variates; evaluate and record:
    \State \quad $\begin{aligned}
    \text{M1:}\quad &lp_t \gets \log\pi_\theta(\mathbf{a}_t \mid \mathbf{s}_t), \quad V_t \gets V_\phi(\mathbf{s}_t),\\
    \text{M2--M4:}\quad &lp_t \gets \log\pi_\theta(\mathbf{a}_t \mid \mathbf{F}_t), \quad V_t \gets V_\phi(\mathbf{F}_t).
    \end{aligned}$
    \State Apply execution threshold: $\widetilde{w}_{i,t} \gets a_{i,t} \cdot \mathbf{1}_{a_{i,t} \ge \tau}$; set executed target weights $\mathbf{w}_t^{\text{target}} \gets \widetilde{\mathbf{w}}_t / \sum_j \widetilde{w}_{j,t}$.
    \State Rebalance: $\text{Turnover}_t \gets \sum_{i=1}^{10} |w_{i,t}^{\text{target}} - w_{i,t}^{\text{cur}}|$, fee $C_t \gets c_{\text{trans}} V_t \text{Turnover}_t$.
    \State Advance environment: update wealth $V_{t+1} \gets (V_t - C_t) \left(\sum_{i=1}^{10} w_{i,t}^{\text{target}}(1+R_{i,t+1}) + w_{\text{cash},t}^{\text{target}}\right)$,
    \State \quad net return $R_{\text{net},t} \gets (V_{t+1}-V_t)/V_t$, and drifted weights $\mathbf{w}_{t+1}^{\text{cur}}$.
    \State Compute model reward: $r_t \gets R_{\text{net},t}$ (M1, M2); $r_t \gets D_t$ (M3); $r_t \gets D_t - \lambda_{\text{turn}}\sum_{i=1}^{10} (w_{i,t}^{\text{target}}-w_{i,t}^{\text{cur}})^2$ (M4).
    \State Store $(\mathbf{F}_t, \mathbf{a}_t, lp_t, V_t, r_t)$ in $\mathcal{B}$ with raw action $\mathbf{a}_t$; $N_{\text{step}} \gets N_{\text{step}} + 1$.
  \EndFor
  \State Standardise rewards $r_t \gets r_t / \sigma_{\text{Welford}}$; compute GAE advantages $\hat{A}_t$ with boundary cuts and returns $\hat{V}_t^{\text{target}}$.
  \For{$\text{epoch} = 1$ \textbf{to} $10$}
    \For{each mini-batch of size $B = 64$ from $\mathcal{B}$}
      \State Evaluate ratio $\rho_i(\theta) \gets \exp(\log \pi_\theta - lp_i)$ and approximate KL divergence; \textbf{break} if $\text{KL} > 0.03$.
      \State Update $(\psi, \theta, \phi)$ via Adam ($\eta = 3 \times 10^{-4}$, grad clip $0.5$), minimising $\mathcal{L}^{\text{total}}(\theta, \phi)$ (Eq.~\eqref{eq:total_loss}).
    \EndFor
  \EndFor
  \If{$N_{\text{step}} \ge N_{\text{eval}}$ \textbf{or} $N_{\text{step}} \ge 300{,}000$}
    \State Evaluate validation Sharpe $\text{SR}_{\text{val}}$; \textbf{if} $\text{SR}_{\text{val}} > \text{SR}_{\text{best}}$ \textbf{then} save checkpoint $\theta^* \gets \theta$, $\text{SR}_{\text{best}} \gets \text{SR}_{\text{val}}$ \textbf{end if}
    \State \textbf{while} $N_{\text{eval}} \le N_{\text{step}}$ \textbf{do} $N_{\text{eval}} \gets N_{\text{eval}} + 25{,}000$ \textbf{end while}
  \EndIf
\EndWhile
\State Restore best checkpoint $\theta^*$ for final out-of-sample test evaluation.
\end{algorithmic}
\end{algorithm}
